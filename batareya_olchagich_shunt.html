Nano kodi (shunt varianti) 
// Batareya o'lchagich (ACS712 siz): Arduino Nano + Bluetooth (HM-10) + shunt rezistor
// Telefon ilovasi (batareya_ilova.html) o'zgarishsiz ishlaydi.
//
// Ulanish (faqat ZARYADSIZLANISH tokini o'lchaydi, "low-side" shunt):
//   Batareya (+)  ->  yuklama (+)
//   Yuklama (-)   ->  shunt rezistor  ->  batareya (-)
//   Nano GND      ->  batareya (-)  (shuntning batareya tomoni)
//   Nano A0       ->  shuntning yuklama tomoni (yuklama (-) bilan bir nuqta)
//   Nano A1       ->  kuchlanish moduli S (0-25V), moduli (-) -> GND, (+) -> batareya (+)
//   Bluetooth TX -> D2, RX -> D3 (1k + 2k bo'luvchi orqali)
//
// Shunt: 0.1 Ohm -> maksimum ~10A (rezistor 10W), 0.05 Ohm -> ~20A.
// Quvvatni hisoblang: P = I*I*R, rezistor quvvati bundan 2 barobar katta bo'lsin.
//
// Buyruqlar: R = Ah/Wh nollash, Z = tok nolini qayta o'rnatish
// Har soniyada yuboriladi:  B,volt,amper,vatt,Ah,Wh

#include <SoftwareSerial.h>

SoftwareSerial bt(2, 3);   // RX, TX

const int CURRENT_PIN = A0;
const int VOLT_PIN    = A1;

const float SHUNT_OHM  = 0.10;   // shunt qarshiligi (Om)
const float VOLT_RATIO = 5.0;    // 25V modul: 5:1
const float VREF_VOLT  = 5.0;    // Nano 5V (multimetr bilan o'lchab to'g'rilang)
const float VREF_INT   = 1.10;   // ichki etalon (har Nano'da 1.0-1.2 oralig'ida, to'g'rilang)

float zeroV = 0;                 // tok nol bo'lgandagi A0 kuchlanishi
float volts = 0, amps = 0, watts = 0, ampHours = 0, wattHours = 0;
unsigned long lastSend = 0;

// Etalonni almashtirib o'qish: almashgandan keyin birinchi o'lchovlar tashlab yuboriladi
float readAvg(int pin, int samples, bool internalRef) {
  analogReference(internalRef ? INTERNAL : DEFAULT);
  delay(5);
  for (int i = 0; i < 5; i++) analogRead(pin);   // tashlab yuboriladi
  long s = 0;
  for (int i = 0; i < samples; i++) s += analogRead(pin);
  float ref = internalRef ? VREF_INT : VREF_VOLT;
  return (s / (float)samples) * ref / 1023.0;
}

void calibrateZero() {
  delay(300);
  zeroV = readAvg(CURRENT_PIN, 500, true);
}

void setup() {
  bt.begin(9600);
  calibrateZero();
  lastSend = millis();
}

void loop() {
  while (bt.available()) {
    char c = bt.read();
    if (c == 'R') { ampHours = 0; wattHours = 0; }
    else if (c == 'Z') calibrateZero();
  }

  unsigned long now = millis();
  if (now - lastSend >= 1000) {
    float dtH = (now - lastSend) / 3600000.0;
    lastSend = now;

    volts = readAvg(VOLT_PIN, 100, false) * VOLT_RATIO;
    float vShunt = readAvg(CURRENT_PIN, 200, true) - zeroV;
    amps = vShunt / SHUNT_OHM;
    if (amps < 0.03) amps = 0;           // shovqinni kesish (faqat zaryadsizlanish)

    watts = volts * amps;
    ampHours  -= amps * dtH;             // ilova: manfiy = zaryadsizlanish
    wattHours -= watts * dtH;

    // Ilova bilan moslik uchun zaryadsizlanishni manfiy tok sifatida yuboramiz
    bt.print("B,");
    bt.print(volts, 2);      bt.print(',');
    bt.print(-amps, 2);      bt.print(',');
    bt.print(-watts, 1);     bt.print(',');
    bt.print(ampHours, 3);   bt.print(',');
    bt.println(wattHours, 2);
  }
}
